# ==============================================================================
# Offline RAG Framework — Requirements
# ==============================================================================
# Python 3.11 is required.
#
# Install all dependencies with:
#   pip install -r requirements.txt
#
# IMPORTANT: This project is designed to run FULLY OFFLINE after initial setup.
# All packages below are installed once and never call external inference APIs.
# ==============================================================================


# ------------------------------------------------------------------------------
# Core RAG Stack
# ------------------------------------------------------------------------------

# ChromaDB — persistent, local vector database for storing embeddings
chromadb

# Sentence Transformers — local embedding model (all-MiniLM-L6-v2)
# Downloads the model once; all subsequent runs are fully offline
sentence-transformers

# Ollama — official Python client for local LLM inference via Ollama
# Communicates with the locally running Ollama server (no cloud calls)
ollama


# ------------------------------------------------------------------------------
# Streamlit — Web UI
# ------------------------------------------------------------------------------

# Streamlit — interactive web application framework
streamlit

# Watchdog — optional but recommended; speeds up Streamlit file watching
watchdog


# ------------------------------------------------------------------------------
# Data Processing
# ------------------------------------------------------------------------------

# Pandas — tabular data loading, cleaning, and stratified sampling
pandas

# NumPy — numerical operations; required by sentence-transformers and chromadb
numpy

# scikit-learn — used for stratified sampling utilities
scikit-learn
